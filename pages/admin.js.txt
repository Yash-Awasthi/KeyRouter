  async function send(path, method, body) {

    const data = await res.json().catch(() => ({ error: { message: `request failed (${res.status})` } }));
    if (!res.ok) throw new Error(data.error?.message || `request failed (${res.status})`);
    return data;
  }

    $("label").value = k.label || "";
    for (const f of NUMBERS) $(f).value = k[f] ?? "";

    $("period").value = k.period || "total";
    $("ips").value = (k.ips || []).join(", ");
    $("downAt").value = k.downgrade?.at ?? "";
    $("downMap").value = (k.downgrade?.map || []).map(([a, b]) => `${a} => ${b}`).join("\n");

  const sumOf = (bucket) => Object.values(bucket || {}).reduce((s, m) => s + (m.cost || 0), 0);
  // The compact cell: one line per key, so a page of 20 stays a page. The detail is a table that opens in place.
  function usage(u) {
    const models = Object.keys(u?.total || {}).length;
    if (!models) return "none";
    const month = u.months?.[thisMonth()] || {};
    const reqs = Object.values(month).reduce((s, m) => s + m.requests, 0);
    return `${usd(sumOf(u.total))} lifetime, ${usd(sumOf(month))} this month (${reqs} req, ${models} model${models === 1 ? "" : "s"})`;
  }

  const modelRow = ([model, m]) => el("tr", {},

    el("td", { className: "num", "data-label": "Input", textContent: m.input.toLocaleString() }),
    el("td", { className: "num", "data-label": "Output", textContent: m.output.toLocaleString() }),

  );

  // The usage table: 20 models at a time, with a real count so nobody has to guess what is hidden.
  const MODELS_PER_PAGE = 20;
  function usageTable(k) {
    const rows = Object.entries(k.usage?.total || {}).sort((a, b) => (b[1].cost || 0) - (a[1].cost || 0));
    const body = el("tbody", {});
    const count = el("span", { className: "muted" });
    const more = el("button", { type: "button", className: "secondary small", textContent: "Show 20 more" });
    let shown = 0;
    const draw = () => {
      const next = rows.slice(shown, shown + MODELS_PER_PAGE);
      shown += next.length;
      body.append(...next.map(modelRow));
      count.textContent = `${shown} of ${rows.length} model${rows.length === 1 ? "" : "s"}, ${usd(sumOf(k.usage?.total))} lifetime`;
      more.hidden = shown >= rows.length;
    };
    more.onclick = draw;
    draw();
    return el("div", { className: "usageDetail" },
      el("div", { className: "head" }, count, more),
      el("table", {}, el("thead", {}, el("tr", {},
        el("th", { textContent: "Model" }), el("th", { className: "num", textContent: "Requests" }),
        el("th", { className: "num", textContent: "Input" }), el("th", { className: "num", textContent: "Output" }),
        el("th", { className: "num", textContent: "Cost" }),

  // A row opens one detail row beneath it; clicking again closes it, so the table never stacks up panels.
  function toggleUsage(k, row) {
    const next = row.nextElementSibling;
    if (next && next.classList.contains("detail")) return next.remove();
    document.querySelectorAll("#list tr.detail").forEach((r) => r.remove());
    const cell = el("td", { colSpan: 6 });
    const models = Object.keys(k.usage?.total || {}).length;
    cell.append(models ? usageTable(k) : el("p", { className: "muted", textContent: "No usage recorded for this key yet." }));
    row.after(el("tr", { className: "detail" }, cell));
  }

  // The search and the state filter are answered by the worker (it can see every key); the cap filter is applied

    const q = $("q").value.trim();
    if (q) params.set("q", q);
    if ($("state").value !== "all") params.set("state", $("state").value);
    return params;

  const syncUrl = () => {
    const params = filterParams();
    const next = location.pathname + (params.toString() ? "?" + params : "");
    if (location.pathname + location.search !== next) history.replaceState(null, "", next);
  };

  // Keyboard: "/" jumps to the search, "n" to a new key, Escape leaves the form and puts focus back on the list.
  // A shortcut must never fire while a field has focus, or typing a label would open forms mid-word.
  const inField = (target) => ["input", "textarea", "select"].includes((target.tagName || "").toLowerCase()) || target.isContentEditable;
  document.addEventListener("keydown", (event) => {
    if (event.key === "Escape") {
      if (document.activeElement === $("q")) return $("q").blur();
      // Escape cancels an edit before it touches a clean form, so it can never clear a half-typed key.
      if (!editing && !inField(event.target)) return;
      resetForm();
      $("formTitle").focus?.();
      return;
    }
    if (inField(event.target) || event.metaKey || event.ctrlKey || event.altKey) return;
    if (event.key === "/") {
      event.preventDefault();
      $("q").focus();
      $("q").select();
    } else if (event.key === "n" || event.key === "N") {
      event.preventDefault();
      resetForm();
      $("label").focus();
      $("label").scrollIntoView({ block: "center" });
    }
  });

  // The whole deployment's report: totals, the models that cost the most, and which keys look unusual.

    const money = (n) => "$" + Number(n).toFixed(n && n < 0.01 ? 6 : 2);
    $("reportTotals").replaceChildren(
      el("span", { textContent: `${report.keys} keys, ${report.totals.requests} requests` }),

    $("reportModels").replaceChildren(...report.top.map((m) => el("tr", {},

    $("reportNote").textContent = report.truncated

  async function loadRotations() {
    const page = await (await fetch("/admin/rotations")).json();
    $("rotationsCard").hidden = !page.pending;

    $("rotations").replaceChildren(...page.rotations.map((r) => {

      const act = async (what) => {
        yes.disabled = no.disabled = true;
        const res = await fetch(`/admin/rotations/${r.id}/${what}`, { method: "POST" });
        const data = await res.json();
        if (!res.ok) { yes.disabled = no.disabled = false; return $("rotateCount").textContent = data.error.message; }
        if (what === "approve") {
          // The new key is shown once, by link, never in this page's own body.

  let batchKeys = [];
  const download = (name, rows) => {

  // Cells starting with = + - @ are prefixed so spreadsheets do not run them as formulas.
  const cell = (v) => {
    let s = String(v);
    if (/^[=+\-@]/.test(s)) s = "'" + s;

  $("export").onclick = async () => {
    const all = [];
    let from = null;
    do {
      const page = await fetchKeys(from);
      all.push(...page.keys);
      from = page.cursor;
    } while (from);
    const rows = [["key", "label", "month", "model", "requests", "input_tokens", "output_tokens", "cost_usd", "priced"]];
    for (const k of all) {
      for (const [month, models] of Object.entries(k.usage.months || {}).sort()) {
        for (const [model, u] of Object.entries(models)) rows.push([k.id, k.label || "", month, model, u.requests, u.input, u.output, u.cost.toFixed(6), u.priced]);
      }
      // Usage recorded before monthly tracking existed only has lifetime totals.
      for (const [model, t] of Object.entries(k.usage.total || {})) {
        const seen = Object.values(k.usage.months || {}).map((m) => m[model]).filter(Boolean);
        const less = (f) => t[f] - seen.reduce((s, m) => s + m[f], 0);
        if (less("requests") > 0) rows.push([k.id, k.label || "", "before monthly tracking", model, less("requests"), less("input"), less("output"), less("cost").toFixed(6), t.priced]);
      }
    }
    download(`key-router-usage-${new Date().toISOString().slice(0, 10)}.csv`, rows);
  };

  // A viewer gets the same page with the writing controls removed: a button that only answers 403 is a trap.
  let viewerMode = false;
  async function loadAuth() {
    const status = await (await fetch("/admin/status")).json();
    viewerMode = status.role === "viewer";
    document.body.classList.toggle("viewer", viewerMode);
    $("viewerNote").hidden = !viewerMode;
    // A viewer may read keys but not the passkey list, so the sign-in card is left to the owner entirely.
    if (viewerMode) {
      $("passkeys").replaceChildren(el("li", { className: "muted", textContent: "Only an owner can see or change passkeys." }));
      load();
      return;
    }
    $("plaintext").hidden = status.encrypted;
    $("passkeyWarning").hidden = status.passkeys > 0 || status.passwordFallback;
    $("signout").hidden = !status.passkeys;
    $("authHint").textContent = status.passkeys
      ? "Passkeys are required to sign in. The admin password no longer works" + (status.passwordFallback ? ", except that ALLOW_PASSWORD is set, so it still does." : ". Keep at least two passkeys so losing one does not lock you out.")
      : "You sign in with the admin password. Add a passkey to sign in with your fingerprint, face or a security key instead; the password then stops working.";
    const list = await (await fetch("/admin/passkeys")).json();
    $("passkeys").replaceChildren(...list.map((p) => el("li", {},
      el("span", { textContent: `${p.name} (added ${new Date(p.created).toLocaleDateString()})` }),
      el("button", {
        className: "ghost",
        textContent: "Remove",
        onclick: async () => {
          if (!confirm(`Remove the passkey "${p.name}"?` + (list.length === 1 ? " The admin password will work again." : ""))) return;
          await fetch("/admin/passkeys/" + p.id, { method: "DELETE" });
          loadAuth();
        },
      }))));
  }
  const b64u = (buf) => btoa(String.fromCharCode(...new Uint8Array(buf))).replaceAll("+", "-").replaceAll("/", "_").replace(/=+$/, "");
  const unb64u = (s) => Uint8Array.from(atob(s.replaceAll("-", "+").replaceAll("_", "/")), (c) => c.charCodeAt(0));
  $("addPasskey").onclick = async () => {
    $("passkeyErr").textContent = "";
    try {
      const o = await send("/admin/passkeys/options", "POST", {});
      const cred = await navigator.credentials.create({ publicKey: {
        challenge: unb64u(o.challenge),
        rp: o.rp,
        user: { id: crypto.getRandomValues(new Uint8Array(16)), name: "admin", displayName: "KeyRouter admin" },
        pubKeyCredParams: [-7, -8, -257].map((alg) => ({ type: "public-key", alg })),
        authenticatorSelection: { residentKey: "required", userVerification: "required" },
        excludeCredentials: o.exclude.map((id) => ({ type: "public-key", id: unb64u(id) })),
        timeout: 60000,
      } });
      const r = cred.response;
      if (!r.getPublicKey) throw new Error("This browser cannot register passkeys here; try a current Chrome, Edge, Safari or Firefox.");
      await send("/admin/passkeys", "POST", {
        id: cred.id,
        clientDataJSON: b64u(r.clientDataJSON),
        publicKey: b64u(r.getPublicKey()),
        alg: r.getPublicKeyAlgorithm(),
        name: $("passkeyName").value.trim() || "Passkey",
      });
      $("passkeyName").value = "";
      loadAuth();
    } catch (e) {
      $("passkeyErr").textContent = e.name === "NotAllowedError" ? "Passkey creation was cancelled or timed out." : e.message;
    }
  };
